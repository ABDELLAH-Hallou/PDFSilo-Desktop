pdfsilo/
├── README.md
├── requirements.txt
├── pdfsilo/
│   ├── __init__.py
│   ├── __main__.py           # enables `python -m pdfsilo`
│   ├── cli.py                # unified argument parser, routes to operations
│   ├── utils.py              # shared helpers (logging, validation, sorting)
│   └── operations/
│       ├── __init__.py
│       ├── concat.py         # merge folder of PDFs → one normalized PDF
│       ├── split.py          # one file per page
│       ├── rotate.py         # rotate all or specific pages
│       ├── extract_range.py  # pull out a page range
│       ├── compress.py       # reduce file size
│       ├── encrypt.py        # AES-256 password protection + permissions
│       ├── decrypt.py        # remove password
│       ├── watermark.py      # stamp text on every page
│       ├── extract_images.py # pull embedded images as PNG/JPEG
│       ├── to_images.py      # render pages as raster images at given DPI
│       └── reorder.py        # rearrange, duplicate, or drop pages
